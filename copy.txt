# Zabbix Agent 2 - zarządzanie na wielu serwerach (serwery sprawdzane po kolei)

# ===== KONFIGURACJA =====
$Servers  = @('SERWER01', 'SERWER02', 'SERWER03')
$Service  = 'Zabbix Agent 2'
$ConfPath = 'C:\Program Files\Zabbix Agent 2\user_parameter2.conf'   # ścieżka na serwerach
$Source   = '\\SHARE\zabbix\user_parameter2.conf'                    # skąd kopiować (opcja 4)
# ========================

function Write-Result($server, $ok, $text) {
    Write-Host ("{0,-25}" -f $server) -NoNewline
    if ($ok) { Write-Host 'OK     ' -ForegroundColor Green -NoNewline } else { Write-Host 'nie ok ' -NoNewline }
    Write-Host $text
}

do {
    Clear-Host
    Write-Host "=== Zabbix Agent 2 ($($Servers.Count) serwerów) ===" -ForegroundColor Cyan
    Write-Host '1 - status usługi'
    Write-Host '2 - restart usługi'
    Write-Host '3 - informacje o pliku (rozmiar, data, SHA256)'
    Write-Host '4 - skopiuj nowy plik'
    Write-Host '0 / exit - zakończ'
    $choice = (Read-Host "`nWybierz opcję").Trim().ToLower()
    Write-Host ''
    $failed = @()

    switch ($choice) {
        '1' {
            foreach ($s in $Servers) {
                try {
                    $st = Invoke-Command -ComputerName $s -ErrorAction Stop -ArgumentList $Service -ScriptBlock {
                        param($n) (Get-Service -Name $n -ErrorAction Stop).Status.ToString()
                    }
                    Write-Result $s ($st -eq 'Running') $st
                } catch { Write-Result $s $false $_.Exception.Message }
            }
        }
        '2' {
            if ((Read-Host "Zrestartować '$Service' na wszystkich serwerach? (T/N)") -notmatch '^(t|tak)$') { break }
            foreach ($s in $Servers) {
                try {
                    $st = Invoke-Command -ComputerName $s -ErrorAction Stop -ArgumentList $Service -ScriptBlock {
                        param($n)
                        Restart-Service -Name $n -Force -ErrorAction Stop
                        (Get-Service -Name $n).Status.ToString()
                    }
                    Write-Result $s ($st -eq 'Running') $st
                    if ($st -ne 'Running') { $failed += $s }
                } catch { Write-Result $s $false $_.Exception.Message; $failed += $s }
            }
            if ($failed) { Write-Host "`nRestart nie powiódł się na: $($failed -join ', ')" }
        }
        '3' {
            foreach ($s in $Servers) {
                try {
                    $info = Invoke-Command -ComputerName $s -ErrorAction Stop -ArgumentList $ConfPath -ScriptBlock {
                        param($p)
                        $f = Get-Item -LiteralPath $p -ErrorAction Stop
                        '{0} B | {1} | {2}' -f $f.Length, $f.LastWriteTime.ToString('yyyy-MM-dd HH:mm:ss'), (Get-FileHash -LiteralPath $p -Algorithm SHA256).Hash
                    }
                    Write-Result $s $true $info
                } catch { Write-Result $s $false $_.Exception.Message }
            }
        }
        '4' {
            if (-not (Test-Path -LiteralPath $Source)) { Write-Host "Nie znaleziono pliku źródłowego: $Source"; break }
            if ((Read-Host "Skopiować $Source na wszystkie serwery? (T/N)") -notmatch '^(t|tak)$') { break }
            foreach ($s in $Servers) {
                try {
                    $sess = New-PSSession -ComputerName $s -ErrorAction Stop
                    Copy-Item -LiteralPath $Source -Destination $ConfPath -ToSession $sess -Force -ErrorAction Stop
                    Write-Result $s $true 'skopiowano'
                } catch { Write-Result $s $false $_.Exception.Message; $failed += $s }
                finally { if ($sess) { Remove-PSSession $sess; $sess = $null } }
            }
            if ($failed) { Write-Host "`nKopiowanie nie powiodło się na: $($failed -join ', ')" }
        }
        { $_ -in '0', 'exit' } { }
        default { Write-Host 'Nieprawidłowy wybór.' }
    }

    if ($choice -notin '0', 'exit') { Read-Host "`nEnter, aby wrócić do menu" | Out-Null }
} while ($choice -notin '0', 'exit')
